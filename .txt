"use client";

import { useEffect, useState, useRef, useCallback } from "react";
import Image from "next/image";
import { AnimatePresence, motion, useReducedMotion } from "motion/react";
import { cn } from "@/src/lib/cn";
import { HERO_SLIDES } from "@/src/data/hero";
import { ABOUT } from "@/src/data/about";
import { Sheet } from "@/src/components/ui/Sheet";
import Link from "next/link";

import { usePreloaderDone } from "@/src/lib/preloader";

const SLIDE_DURATION = 6500;
const SWIPE_THRESHOLD = 40;

function AboutSheet({
  isOpen,
  onClose,
}: {
  isOpen: boolean;
  onClose: () => void;
}) {
  return (
    <Sheet isOpen={isOpen} onClose={onClose} title="Our story">
      <div className="space-y-8 text-grey-dark">
        <div>
          <h2 className="display text-3xl font-bold tracking-tight mb-4">
            About Ranchi Security
          </h2>
          <p className="text-[14.5px] leading-relaxed mb-4">{ABOUT.body}</p>
          <p className="text-[14.5px] leading-relaxed">
            Since 1995 we have guarded industry, mines and institutions across
            Jharkhand, Bihar and Odisha, and trained 25,000 youth through Skill
            India and MEPSC programmes.
          </p>
        </div>

        <div className="py-4 border-y border-grey-soft">
          <p className="text-accent font-medium text-sm">
            Grew 5x in a year and 10x in 18 months, deploying 500+ ex-servicemen.
          </p>
        </div>

        <div>
          <h3 className="text-xs font-bold uppercase tracking-wider text-grey-mid mb-4">
            {ABOUT.groupTitle}
          </h3>
          <ul className="space-y-4">
            {ABOUT.organisations.map((org, i) => (
              <li key={i}>
                <p className="font-semibold text-sm">{org.name}</p>
                <p className="text-xs text-grey-mid mt-0.5">{org.note}</p>
              </li>
            ))}
          </ul>
        </div>

        <div className="pt-2">
          <button
            onClick={onClose}
            className="flex w-full items-center justify-center rounded-sm bg-accent px-6 py-4 text-sm font-semibold tracking-wide text-white transition-colors hover:bg-accent-hover"
          >
            Close
          </button>
        </div>
      </div>
    </Sheet>
  );
}

export function Hero() {
  const preloaderDone = usePreloaderDone();
  const [active, setActive] = useState(0);
  const [isPaused, setIsPaused] = useState(false);
  const [aboutOpen, setAboutOpen] = useState(false);
  const reduce = useReducedMotion();
  const touchStartX = useRef<number | null>(null);
  const heroRef = useRef<HTMLDivElement>(null);

  const nextSlide = useCallback(() => {
    setActive((prev) => (prev + 1) % HERO_SLIDES.length);
  }, []);

  const prevSlide = useCallback(() => {
    setActive((prev) => (prev - 1 + HERO_SLIDES.length) % HERO_SLIDES.length);
  }, []);

  // Keyboard navigation
  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      if (
        document.activeElement === document.body ||
        heroRef.current?.contains(document.activeElement)
      ) {
        if (e.key === "ArrowRight") {
          e.preventDefault();
          nextSlide();
        } else if (e.key === "ArrowLeft") {
          e.preventDefault();
          prevSlide();
        }
      }
    };
    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, [nextSlide, prevSlide]);

  // Autoplay
  useEffect(() => {
    if (isPaused || !preloaderDone) return;
    const timer = setTimeout(nextSlide, SLIDE_DURATION);
    return () => clearTimeout(timer);
  }, [active, isPaused, nextSlide, preloaderDone]);

  // Touch Swipe
  const onTouchStart = (e: React.TouchEvent) => {
    touchStartX.current = e.touches[0].clientX;
  };
  const onTouchEnd = (e: React.TouchEvent) => {
    if (touchStartX.current === null) return;
    const dx = e.changedTouches[0].clientX - touchStartX.current;
    if (Math.abs(dx) > SWIPE_THRESHOLD) {
      if (dx < 0) nextSlide();
      else prevSlide();
    }
    touchStartX.current = null;
  };

  const current = HERO_SLIDES[active];

  return (
    <>
      <section
        id="home"
        data-tone="dark"
        ref={heroRef}
        aria-label="Hero"
        className="relative flex flex-col justify-between h-[110svh] max-h-[110svh] w-full bg-[#0F0E0D] text-white overflow-hidden"
        onMouseEnter={() => setIsPaused(true)}
        onMouseLeave={() => setIsPaused(false)}
        tabIndex={-1}
      >
        {/* 1. BACKGROUND IMAGE SLIDER (78% WIDTH ON DESKTOP FOR PERFECT IMAGE SHOWCASE) */}
        <div
          className="absolute inset-0 h-full w-full lg:left-[5%] lg:w-[95%] pointer-events-auto overflow-hidden z-0"
          onTouchStart={onTouchStart}
          onTouchEnd={onTouchEnd}
        >
          {HERO_SLIDES.map((slide, i) => {
            const isActive = i === active;

            return (
              <div
                key={i}
                className={cn(
                  "absolute inset-0 w-full h-full transition-opacity duration-1000 ease-in-out",
                  isActive ? "opacity-100 z-10" : "opacity-0 pointer-events-none z-0"
                )}
                aria-hidden={!isActive}
              >
                <div
                  className={cn(
                    "relative w-full h-full transform-gpu",
                    isActive && !reduce && preloaderDone && "animate-[heroZoom_8s_ease-out_forwards]"
                  )}
                >
                  <Image
                    src={slide.src}
                    alt={slide.alt}
                    fill
                    priority={i === 0}
                    className="object-cover object-center w-full h-full"
                    style={{
                      objectPosition: slide.focal || "center",
                    }}
                    sizes="(max-width: 1024px) 100vw, 78vw"
                  />
                </div>
              </div>
            );
          })}

          <style
            dangerouslySetInnerHTML={{
              __html: `
            @keyframes heroZoom {
              from { transform: scale(1.05); }
              to { transform: scale(1); }
            }
          `,
            }}
          />

          {/* Desktop Mask Gradient (Flows smoothly from dark canvas into the photo) */}
          <div className="hidden lg:block absolute inset-y-0 left-0 w-2/5 z-20 pointer-events-none bg-gradient-to-r from-[#0F0E0D] via-[#0F0E0D]/60 to-transparent" />

          {/* Mobile Overlay Scrim */}
          <div className="block lg:hidden absolute inset-0 z-20 pointer-events-none bg-gradient-to-b from-[#0F0E0D]/80 via-black/30 to-[#0F0E0D]/60" />
        </div>

        {/* 2. HERO CONTENT AREA */}
        <div className="relative z-20  max-w-[1400px] mx-auto w-full px-6 sm:px-10 lg:px-16 pt-8 sm:pt-10 lg:pt-12 pb-6 pointer-events-none">
          <div className="w-full max-w-xl pointer-events-auto">
            {/* Optimized Headline */}
            <h1
              className="text-white pt-70 font-semibold text-[clamp(1.9rem,3.8vw,3.1rem)] leading-[1.12] tracking-tight mb-4 sm:mb-5 drop-shadow-[0_4px_16px_rgba(0,0,0,0.95)]"
              style={{ fontFamily: "'Newsreader', serif" }}
            >
              Security built on{" "}
              <em className="not-italic text-[#FF8595] italic font-normal">
                discipline
              </em>
              , delivered by ex-servicemen.
            </h1>

            {/* Subtext */}
            <p
              className="text-sm sm:text-base lg:text-lg leading-relaxed text-white/90 font-normal mb-6 sm:mb-8 max-w-[480px] drop-shadow-[0_2px_8px_rgba(0,0,0,0.95)]"
              style={{ fontFamily: "'Inter', sans-serif" }}
            >
              Guarding industry, mines and institutions across Jharkhand, Bihar and
              Odisha — backed by over 25,000 certified personnel.
            </p>

            {/* CTA Buttons */}
            {/* <div className="flex flex-col sm:flex-row items-center gap-3.5 w-full sm:w-auto">
              <Link
                href="#contact"
                className="w-full sm:w-auto inline-flex items-center justify-center gap-2 px-8 h-[48px] rounded-md bg-[#9E002B] hover:bg-[#B80032] text-xs sm:text-sm font-bold tracking-[0.1em] uppercase text-white transition-all shadow-xl hover:shadow-red-900/40 active:scale-95"
              >
                Request a Quote
                <svg
                  className="w-4 h-4"
                  fill="none"
                  stroke="currentColor"
                  viewBox="0 0 24 24"
                >
                  <path
                    strokeLinecap="round"
                    strokeLinejoin="round"
                    strokeWidth="2.5"
                    d="M14 5l7 7m0 0l-7 7m7-7H3"
                  />
                </svg>
              </Link>

              <button
                type="button"
                onClick={() => setAboutOpen(true)}
                className="w-full sm:w-auto inline-flex items-center justify-center h-[48px] px-8 rounded-md border border-white/35 bg-black/30 hover:bg-white/15 text-xs sm:text-sm font-bold tracking-[0.1em] uppercase text-white transition-all backdrop-blur-sm active:scale-95"
              >
                Our Story
              </button>
            </div> */}
          </div>
        </div>

        {/* 3. BOTTOM SLIDER CONTROLS BAR */}
        <div className="relative z-30 w-full bg-gradient-to-r from-transparent via-[#6b001a]/20 to-[#6b001a]/75 backdrop-blur-sm lg:backdrop-blur-none border-t border-white/10 pointer-events-auto">
          <div className="max-w-[1400px] mx-auto px-6 sm:px-10 lg:px-16 h-16 sm:h-20 flex items-center justify-between">
            {/* Left Caption / Slide Title (Desktop Only) */}
            <div className="hidden lg:flex items-center gap-3">
              <span className="w-2 h-2 rounded-full bg-[#FF8595] animate-ping" />
              <AnimatePresence mode="wait">
                <motion.span
                  key={active}
                  initial={{ opacity: 0, x: -10 }}
                  animate={{ opacity: 1, x: 0 }}
                  exit={{ opacity: 0, x: 10 }}
                  className="text-xs sm:text-sm font-medium tracking-wide text-white/95 drop-shadow-md"
                >
                  {current.caption}
                </motion.span>
              </AnimatePresence>
            </div>

            {/* Right Side: Slide Thumbnails */}
            <div className="flex items-center justify-between w-full lg:w-auto gap-4 sm:gap-6">
              {/* Counter Indicator */}
              {/* <span className="text-xs font-mono font-semibold text-white/80 tracking-widest">
                {String(active + 1).padStart(2, "0")} /{" "}
                {String(HERO_SLIDES.length).padStart(2, "0")}
              </span> */}

              {/* Slide Thumbnails */}
              <div className="flex gap-2 sm:gap-3 items-center">
                {HERO_SLIDES.map((slide, i) => (
                  <button
                    key={i}
                    onClick={() => setActive(i)}
                    className="relative focus:outline-none"
                    aria-label={`Slide ${i + 1}`}
                  >
                    <div
                      className={cn(
                        "relative w-12 h-8 sm:w-16 sm:h-10 rounded overflow-hidden transition-all duration-300",
                        active === i
                          ? "ring-2 ring-white opacity-100 scale-105 shadow-lg"
                          : "opacity-40 hover:opacity-90 border border-white/20"
                      )}
                    >
                      <Image
                        src={slide.src}
                        alt=""
                        fill
                        sizes="64px"
                        className="object-cover"
                        style={{ objectPosition: slide.focal || "center" }}
                      />
                    </div>

                    {/* Active Progress Track Line */}
                    {active === i && (
                      <div className="absolute -bottom-1 left-0 right-0 h-[2px] bg-white/30 overflow-hidden rounded-full">
                        <motion.div
                          className="h-full bg-white origin-left"
                          initial={{ scaleX: 0 }}
                          animate={{ scaleX: isPaused ? 0 : 1 }}
                          transition={{
                            duration: SLIDE_DURATION / 1000,
                            ease: "linear",
                          }}
                        />
                      </div>
                    )}
                  </button>
                ))}
              </div>
            </div>
          </div>
        </div>
      </section>

      <AboutSheet isOpen={aboutOpen} onClose={() => setAboutOpen(false)} />
    </>
  );
}
